### Establish Connection
~~~
!@NXOS-1  LACP
conf t
 int e1/1-4
  no shut
  channel-group 10 mode active
  exit
  !
 int po10
  no shut
  no switchport
  mtu 9216
  ip add 100.1.1.1/30
  end
~~~

~~~
!@NXOS-2  LACP
conf t
 int e1/1-4
  no shut
  channel-group 10 mode passive
  exit
  !
 int po10
  no shut
  no switchport
  mtu 9216
  ip add 100.1.1.2/30
  end
~~~

### VXLAN

1. VXLAN - Virtual Extensible LAN
2. VNI - VXLN Network Identifier
3. VN-SEGMENT - Virtual Network Segment
4. NVE - Network Virtualization Endpoint
5. VTEP - VXLAN Tunnel Endpoint


~~~
!@NXOS-1
conf t
 vlan 50
  name MANILA-TOKYO
  vn-segment 5000
  exit
 !
 int e1/5
  no shut
  switchport
  switchport access vlan 50
  exit
 !
 int nve 1
  no shut
  source-interface lo0
  member vni 5000
   ingress-replication protocol static
    peer-ip 20.20.20.20
	end
show vxlan
show nve peers
show nve vni
~~~
